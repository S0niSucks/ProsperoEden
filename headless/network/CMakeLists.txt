# This directory also builds independently for focused host/SDK transport checks.
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
    cmake_minimum_required(VERSION 3.24)
    project(prosperoeden_network LANGUAGES C CXX)
    set(CMAKE_CXX_STANDARD 20)
endif()
set(EDEN_NETWORK_DEPS "${CMAKE_CURRENT_LIST_DIR}/../../.deps" CACHE PATH "Pinned network client sources (tools/deps.json)")
function(network_clients)
    set(CMAKE_POLICY_DEFAULT_CMP0077 NEW)
    set(CMAKE_POLICY_DEFAULT_CMP0126 NEW)
    set(BUILD_SHARED_LIBS OFF)
    set(ENABLE_EXAMPLES OFF)
    set(ENABLE_TESTS OFF)
    set(ENABLE_UTILS OFF)
    set(ENABLE_DOCUMENTATION OFF)
    set(ENABLE_LIBKRB5 OFF)
    set(ENABLE_GSSAPI OFF)
    set(ENABLE_TLS OFF)
    set(ENABLE_NFS4_2 OFF)
    set(ENABLE_MULTITHREADING OFF)
    # configure probes must compile against the SDK, never infer Linux host APIs when cross-compiling.
    foreach(pair "libsmb2-libsmb2-6.3|smb2" "libnfs-libnfs-8.0.0|nfs")
        string(REPLACE "|" ";" parts "${pair}")
        list(GET parts 0 source)
        list(GET parts 1 binary)
        if(NOT EXISTS "${EDEN_NETWORK_DEPS}/${source}/CMakeLists.txt")
            message(FATAL_ERROR "Missing pinned ${source}; run python3 tools/deps.py fetch (no system SMB/NFS fallback)")
        endif()
        add_subdirectory("${EDEN_NETWORK_DEPS}/${source}" "${CMAKE_CURRENT_BINARY_DIR}/${binary}" EXCLUDE_FROM_ALL)
    endforeach()
endfunction()
network_clients()
add_library(eden-network STATIC transport.cpp network.cpp)
target_include_directories(eden-network PUBLIC "${CMAKE_CURRENT_LIST_DIR}/..")
target_include_directories(eden-network PRIVATE "${EDEN_NETWORK_DEPS}/libsmb2-libsmb2-6.3/include"
    "${EDEN_NETWORK_DEPS}/libnfs-libnfs-8.0.0/include")
target_link_libraries(eden-network PUBLIC smb2 nfs)
if(TARGET nlohmann_json::nlohmann_json)
    target_link_libraries(eden-network PRIVATE nlohmann_json::nlohmann_json)
endif()
if(EDEN_NETWORK_CHECK)
    add_executable(eden-network-check check.cpp)
    target_link_libraries(eden-network-check PRIVATE eden-network pthread)
    target_compile_options(eden-network-check PRIVATE -UNDEBUG)
endif()
